set(WRAP_FILE LBMPluginWrapper.cpp)
set(MODULE_NAME openmmlbm)

# SWIG modules exchange types (an LBMForce passed to System.addForce()) only if they were generated
# by the same SWIG version: check it against the one recorded in the OpenMM Python module.
execute_process(
    COMMAND "${PYTHON_EXECUTABLE}" -c "import openmm, os; print(os.path.join(os.path.dirname(openmm.__file__), 'openmm.py'))"
    OUTPUT_VARIABLE OPENMM_PY_FILE OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND "${SWIG_EXECUTABLE}" -version OUTPUT_VARIABLE SWIG_VERSION_OUTPUT)
string(REGEX MATCH "Version ([0-9]+\\.[0-9]+)" _match "${SWIG_VERSION_OUTPUT}")
set(LBM_SWIG_VERSION "${CMAKE_MATCH_1}")
IF(EXISTS "${OPENMM_PY_FILE}")
    file(STRINGS "${OPENMM_PY_FILE}" OPENMM_SWIG_LINE REGEX "Version [0-9]+\\.[0-9]+" LIMIT_COUNT 1)
    string(REGEX MATCH "Version ([0-9]+\\.[0-9]+)" _match "${OPENMM_SWIG_LINE}")
    set(OPENMM_SWIG_VERSION "${CMAKE_MATCH_1}")
    IF(NOT LBM_SWIG_VERSION STREQUAL OPENMM_SWIG_VERSION)
        MESSAGE(FATAL_ERROR "The OpenMM Python module was generated by SWIG ${OPENMM_SWIG_VERSION}, but ${SWIG_EXECUTABLE} is SWIG ${LBM_SWIG_VERSION}: install SWIG ${OPENMM_SWIG_VERSION}, or disable the Python wrapper with -DLBM_BUILD_PYTHON_WRAPPERS=OFF")
    ENDIF()
ELSE()
    MESSAGE(WARNING "Could not find the OpenMM Python module: the SWIG version cannot be checked")
ENDIF()

# Execute SWIG to generate source code for the Python module.

add_custom_command(
    OUTPUT "${WRAP_FILE}"
    COMMAND "${SWIG_EXECUTABLE}"
        -python -c++
        -o "${WRAP_FILE}"
        "-I${OPENMM_DIR}/include"
        "${CMAKE_CURRENT_SOURCE_DIR}/openmmlbm.i"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/openmmlbm.i" "${SWIG_EXECUTABLE}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
)

# Compile the Python module.

add_custom_target(PythonInstall DEPENDS "${WRAP_FILE}" "${CMAKE_CURRENT_SOURCE_DIR}/setup.py")
add_dependencies(PythonInstall ${SHARED_LBM_TARGET})
set(LBM_PLUGIN_HEADER_DIR "${CMAKE_SOURCE_DIR}/openmmapi/include")
IF(WIN32)
    set(LBM_PLUGIN_LIBRARY "${CMAKE_BINARY_DIR}/${CMAKE_IMPORT_LIBRARY_PREFIX}${SHARED_LBM_TARGET}${CMAKE_IMPORT_LIBRARY_SUFFIX}")
ELSE()
    set(LBM_PLUGIN_LIBRARY "${CMAKE_BINARY_DIR}/${CMAKE_SHARED_LIBRARY_PREFIX}${SHARED_LBM_TARGET}${CMAKE_SHARED_LIBRARY_SUFFIX}")
ENDIF()
configure_file(${CMAKE_CURRENT_SOURCE_DIR}/setup.py ${CMAKE_CURRENT_BINARY_DIR}/setup.py)
add_custom_command(TARGET PythonInstall POST_BUILD
    COMMAND "${PYTHON_EXECUTABLE}" -m pip install --no-deps --no-build-isolation .
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
)
